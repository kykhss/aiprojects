? How to setup your Google Sheet (Easy 3 Steps)

  1. Step 1: Open Google Sheets & Apps Script Create a new Google Sheet at sheets.new. In the top menu, click Extensions → Apps Script.
  2. Step 2: Paste the Apps Script Code

    Delete any existing code in Code.gs and paste the snippet below:

    function doGet(e) {
  const action = e.parameter.action;
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getActiveSheet();
  
  if (action === "read") {
    const data = sheet.getDataRange().getValues();
    if (data.length <= 1) {
      return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
    }
    const headers = data.shift(); // remove header
    const rows = data.map(function(r) {
      let d = r[0];
      if (d instanceof Date) {
        d = Utilities.formatDate(d, Session.getScriptTimeZone(), "yyyy-MM-dd");
      }
      return {
        date: String(d || "").trim(),
        className: String(r[1] || "").trim(),
        division: String(r[2] || "").trim(),
        hours: Number(r[3]) || 0,
        remarks: String(r[4] || "").trim()
      };
    }).filter(function(item) { return item.date !== ""; });
    
    return ContentService.createTextOutput(JSON.stringify(rows)).setMimeType(ContentService.MimeType.JSON);
  }
  return ContentService.createTextOutput(JSON.stringify({ status: "active" })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sheet = ss.getActiveSheet();
    
    // Ensure header row exists
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Date", "Class", "Division", "Hours", "Remarks", "Timestamp"]);
    }
    
    const body = JSON.parse(e.postData.contents);
    sheet.appendRow([
      body.date,
      body.className,
      body.division,
      Number(body.hours),
      body.remarks || "",
      new Date().toISOString()
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}
  3. Step 3: Deploy as Web App

    In Apps Script, click Deploy > New deployment.

    • Click gear icon & select Web app
    • Execute as: Me (your email)
    • Who has access: Anyone (crucial for form submissions)
